import re, sys
src = sys.argv[1] if len(sys.argv) > 1 else "puzzle_netlist.txt"
dst = sys.argv[2] if len(sys.argv) > 2 else "puzzle.v"
txt = open(src).read()
body = re.search(r'^circuit puzzle \(\);\n(.*?)\n^end;', txt, re.S|re.M).group(1)
insts = re.findall(r'^\s*subcircuit (\S+) (\S+) \((.*)\);$', body, re.M)

PHYS  = ('tapvpwrvgnd','decap','diode','fill','conb_1')   # no logical function
PWR   = {'VPWR','VGND','VPB','VNB'}
PORTS = {'I','clk','rst_n','enable','success'} | {f"O[{i}]" for i in range(8)}

def split_pins(s): return re.findall(r"(?:'[^']*'|[^,])+", s)
def parts(t):      return t.strip().strip("'").split(',')

def netname(tok):
    p = parts(tok)
    if 'VPWR' in p: return "1'b1"          # incl. tie-high 'HI,VPWR'
    if 'VGND' in p: return "1'b0"          # incl. tie-low  'LO,VGND'
    n = p[0]
    return n if n in PORTS else 'n' + n.lstrip('$').replace('I', 'i')

wires, lines = set(), []
for cell, inst, pins in insts:
    if cell.startswith('VIA_') or any(k in cell for k in PHYS): continue
    conns = []
    for tok in split_pins(pins):
        pin, net = tok.split('=', 1)
        pin = parts(pin)[0]
        if pin in PWR: continue
        w = netname(net)
        if not w.startswith("1'b") and w not in PORTS: wires.add(w)
        conns.append(f".{pin}({w})")
    lines.append(f"  {cell} u{inst.lstrip('$')} ({', '.join(conns)});")

with open(dst, 'w') as f:
    f.write("module puzzle (\n  input I, clk, rst_n, enable,\n"
            "  output success,\n  output [7:0] O\n);\n")
    for w in sorted(wires): f.write(f"  wire {w};\n")
    f.write("\n" + "\n".join(lines) + "\nendmodule\n")
print(f"{len(lines)} instances, {len(wires)} wires -> {dst}")
